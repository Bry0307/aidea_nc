        'gcash_ref',
        'service',
        'service_id',
        'student',
        'student_id',
        'date',
        'date_iso',
        'amount',
        'method',
        'status',
        'proof_image',
        'research_items',
    ];

    public function user()
    {
        return $this->belongsTo(User::class);
    }
