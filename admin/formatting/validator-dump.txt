447:     return new Promise((resolve, reject) => {
448:    ~_validatorDeps.ensurePdfJs(async () => {

454:    ~const paragraphs = [];
455:    ~const marginSamples = { left: [], right: [], top: [], bottom: [] };
456: 
457:    ~for (let pageNum = 1; pageNum <= maxPages; pageNum++) {
458:    ~const page = await pdf.getPage(pageNum);
459:    ~const viewport = page.getViewport({ scale: 1 });
460:    ~const pageWidthIn = viewport.width / _POINTS_PER_INCH;
461:    ~const pageHeightIn = viewport.height / _POINTS_PER_INCH;
462:    ~const content = await page.getTextContent();
463: 
464:    ~const items = content.items
465:    ~.filter(it => it.str && it.str.trim())
466:    ~.map(it => ({
467:    ~str: it.str,
468:    ~x: it.transform[4],
469:    ~y: it.transform[5],
470:    ~fontSize: Math.hypot(it.transform[2], it.transform[3]) || it.height || 10,
471:    ~}));
472:    ~if (!items.length) continue;
473: 
474:    ~items.sort((a, b) => b.y - a.y || a.x - b.x);
475:    ~const lines = [];
476:    ~let current = null;
477:    ~const Y_TOL = 2;
478:    ~items.forEach(it => {
479:    ~if (current && Math.abs(current.y - it.y) <= Y_TOL) {
480:    ~current.items.push(it);
481:    ~} else {
482:    ~current = { y: it.y, items: [it] };
483:    ~lines.push(current);
484:    ~}
485:    ~});
486: 
487:    ~const xs = items.map(it => it.x);
488:    ~const ys = items.map(it => it.y);
489:    ~marginSamples.left.push(Math.min(...xs) / _POINTS_PER_INCH);
490:    ~marginSamples.right.push(pageWidthIn - Math.max(...xs) / _POINTS_PER_INCH);
491:    ~marginSamples.top.push(pageHeightIn - Math.max(...ys) / _POINTS_PER_INCH);
492:    ~marginSamples.bottom.push(Math.min(...ys) / _POINTS_PER_INCH);
493: 
494:    ~for (let li = 1; li < lines.length; li++) {
495:    ~const prev = lines[li - 1], cur = lines[li];
496:    ~const fontSize = cur.items[0].fontSize || 10;
497:    ~const deltaY = Math.abs(prev.y - cur.y);
498:    ~const lineSpacing = fontSize ? +(deltaY / fontSize).toFixed(2) : null;
499:    ~const lineStartX = cur.items[0].x;
500:    ~const text = cur.items.map(it => it.str).join('').trim();
501:    ~if (!text) continue;
502:    ~paragraphs.push({
503:    ~index: paragraphs.length,
504:    ~page: pageNum,
505:    ~y: cur.y,
506:    ~height: fontSize,
507:    ~snippet: text.slice(0, 70) + (text.length > 70 ? '?' : ''),
508:    ~text,
509:    ~lineSpacing,
510:    ~lineStartX,
511:    ~});
512:    ~}
513:    ~}
514: 
515:    ~const avg = arr => arr.length ? +(arr.reduce((a, b) => a + b, 0) / arr.length).toFixed(2) : null;
516:    ~const margins = {
517:    ~left: avg(marginSamples.left), right: avg(marginSamples.right),
518:    ~top: avg(marginSamples.top), bottom: avg(marginSamples.bottom),
519:    ~};
520: 
521:    ~const typicalLeftX = margins.left != null ? margins.left * _POINTS_PER_INCH : null;
522:    ~paragraphs.forEach(p => {
523:    ~if (typicalLeftX != null && p.lineStartX != null) {
524:    ~const deltaIn = (p.lineStartX - typicalLeftX) / _POINTS_PER_INCH;
525:    ~p.firstLineIndent = deltaIn > 0.15 ? +deltaIn.toFixed(2) : 0;
526:    ~}
527:    ~});
528: 
529:    ~resolve({ source: 'pdf', estimated: true, margins, paragraphs });
530:    ~} catch (e) {
531:    ~reject(e);
532:    ~}
533:    ~});
534:     });

545: function _evaluateFormattingRule(rule, profile) {
546:     if (!profile) return null;
547: 
548:     const type = (rule.type || '').toLowerCase();
549:     const tolerance = typeof rule.tolerance === 'number' ? rule.tolerance : _RULE_TOLERANCE_DEFAULTS[type];
550: 
551:     if (type === 'spacing') {
552:    ~const target = parseFloat(rule.value);
553:    ~if (Number.isNaN(target)) return null;
554:    ~const checked = profile.paragraphs.filter(p => p.lineSpacing != null);
555:    ~if (!checked.length) return { status: 'manual', violations: [], checkedCount: 0 };
556:    ~const violations = checked
557:    ~.filter(p => Math.abs(p.lineSpacing - target) > tolerance)
558:    ~.map(p => ({ ...p, actual: p.lineSpacing }));
559:    ~return { status: violations.length ? 'fail' : 'pass', violations, checkedCount: checked.length, target, unit: 'x line height' };
560:     }
561: 
562:     if (type === 'indentation') {
563:    ~const target = parseFloat(rule.value);
564:    ~if (Number.isNaN(target)) return null;
565:    ~const checked = profile.paragraphs.filter(p => p.firstLineIndent != null);
566:    ~if (!checked.length) return { status: 'manual', violations: [], checkedCount: 0 };
567:    ~const violations = checked
568:    ~.filter(p => Math.abs(p.firstLineIndent - target) > tolerance)
569:    ~.map(p => ({ ...p, actual: p.firstLineIndent }));
570:    ~return { status: violations.length ? 'fail' : 'pass', violations, checkedCount: checked.length, target, unit: 'in' };
571:     }
572: 
573:     if (type === 'alignment') {
574:    ~const target = _alignNormalize(rule.value);
575:    ~if (!target) return null;
576:    ~const checked = profile.paragraphs.filter(p => p.alignment);
577:    ~if (!checked.length) return { status: 'manual', violations: [], checkedCount: 0 };
578:    ~const violations = checked
579:    ~.filter(p => p.alignment !== target)
580:    ~.map(p => ({ ...p, actual: p.alignment }));
581:    ~return { status: violations.length ? 'fail' : 'pass', violations, checkedCount: checked.length, target };
582:     }
583: 
584:     if (type === 'margin') {
585:    ~if (!profile.margins) return { status: 'manual', violations: [], checkedCount: 0 };
586:    ~const targets = (rule.value && typeof rule.value === 'object')
587:    ~? rule.value
588:    ~: { top: rule.value, bottom: rule.value, left: rule.value, right: rule.value };
589:    ~const violations = [];
590:    ~['top', 'bottom', 'left', 'right'].forEach(side => {
591:    ~const t = parseFloat(targets[side]);
592:    ~const actual = profile.margins[side];
593:    ~if (Number.isNaN(t) || actual == null) return;
594:    ~if (Math.abs(actual - t) > tolerance) {
595:    ~violations.push({ side, target: t, actual: +actual.toFixed(2) });
596:    ~}
597:    ~});
598:    ~return { status: violations.length ? 'fail' : 'pass', violations, checkedCount: 4, isMargin: true };
599:     }
600: 
601:     return null;
602: }
