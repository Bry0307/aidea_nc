875: function _dvDrawLine(page, font, words, x0, y, size, usableWidth, indent, align, isLastLine, color) {
876:     const lineText = words.join(' ');
877:     const naturalWidth = font.widthOfTextAtSize(lineText, size);
878:     const avail = usableWidth - indent;
879: 
880:     if (align === 'center') {
881:    ~const offset = Math.max(0, (avail - naturalWidth) / 2);
882:    ~page.drawText(lineText, { x: x0 + indent + offset, y, size, font, color });
883:     } else if (align === 'right') {
884:    ~const offset = Math.max(0, avail - naturalWidth);
885:    ~page.drawText(lineText, { x: x0 + indent + offset, y, size, font, color });
886:     } else if (align === 'justified' && !isLastLine && words.length > 1) {
887:    ~const spaceWidth = font.widthOfTextAtSize(' ', size);
888:    ~const wordsWidth = words.reduce((sum, w) => sum + font.widthOfTextAtSize(w, size), 0);
889:    ~const gaps = words.length - 1;
890:    ~const extra = Math.max(0, avail - wordsWidth - spaceWidth * gaps);
891:    ~const gapWidth = spaceWidth + extra / gaps;
892:    ~let x = x0 + indent;
893:    ~words.forEach(w => {
894:    ~page.drawText(w, { x, y, size, font, color });
895:    ~x += font.widthOfTextAtSize(w, size) + gapWidth;
896:    ~});
897:     } else {
898:    ~page.drawText(lineText, { x: x0 + indent, y, size, font, color });
899:     }
